cmake_minimum_required(VERSION 3.24)
project(msf_tools_native VERSION 0.1.0 LANGUAGES CXX)
option(MSF_BUILD_NATIVE_PILOT "Build the bounded numerical pilot" ON)
option(MSF_BUILD_PYTHON_BINDINGS "Build optional pybind11 numerical bindings" OFF)
if(MSF_BUILD_NATIVE_PILOT)
  add_library(msf_common native/src/common.cpp)
  add_library(msf_expression native/src/expression.cpp)
  add_library(msf_skeletal native/src/skeletal.cpp)
  foreach(component common expression skeletal)
    target_compile_features(msf_${component} PUBLIC cxx_std_17)
    target_include_directories(msf_${component} PUBLIC native/include)
    set_target_properties(msf_${component} PROPERTIES POSITION_INDEPENDENT_CODE ON)
    if(MSVC)
      target_compile_options(msf_${component} PRIVATE /W4)
    else()
      target_compile_options(msf_${component} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
  endforeach()
  target_link_libraries(msf_expression PUBLIC msf_common)
  target_link_libraries(msf_skeletal PUBLIC msf_common)
  enable_testing()
  add_executable(msf_native_tests native/tests/test_native.cpp)
  target_link_libraries(msf_native_tests PRIVATE msf_expression msf_skeletal)
  add_test(NAME msf_native_tests COMMAND msf_native_tests)
  if(MSF_BUILD_PYTHON_BINDINGS)
    find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
    find_package(pybind11 3.1 CONFIG REQUIRED)
    pybind11_add_module(_msf_native native/src/bindings.cpp)
    target_link_libraries(_msf_native PRIVATE msf_expression msf_skeletal)
    add_test(NAME msf_python_native_agreement
      COMMAND ${Python_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/native/tests/test_agreement.py
              $<TARGET_FILE_DIR:_msf_native> ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME msf_python_native_strict
      COMMAND ${Python_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/native/tests/test_strict.py
              $<TARGET_FILE_DIR:_msf_native> ${CMAKE_CURRENT_SOURCE_DIR})
  endif()
endif()
